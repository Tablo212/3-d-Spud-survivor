    weaponMeshes.push(mesh);
  });
}

// --- SPAWNING ENEMIES (AGGRESSIVE & DEADLY) ---
let spawnTimer = 0;
function spawnEnemy() {
  const angle = Math.random() * Math.PI * 2;
  const dist = CONFIG.arenaSize - 2;
  const x = Math.cos(angle) * dist;
  const z = Math.sin(angle) * dist;

  // Enemy types unlock/increase with waves
  const roll = Math.random();
  let type = 'rusher';
  if (state.wave >= 2 && roll > 0.65) type = 'shooter';
  if (state.wave >= 1 && roll > 0.82) type = 'brute';
  if (roll < 0.25) type = 'sprinter';

  const waveMultHp = Math.pow(CONFIG.enemyHpScale, state.wave - 1);
  const waveMultDmg = Math.pow(CONFIG.enemyDmgScale, state.wave - 1);

  let cfg = {
    rusher:   { hp: 28, speed: 8.5, dmg: 12, size: 1.0, color: 0xff4757, xp: 5, mat: 2 },
    sprinter: { hp: 16, speed: 13.5, dmg: 10, size: 0.75, color: 0xffa502, xp: 6, mat: 2 },
    shooter:  { hp: 35, speed: 5.5, dmg: 14, size: 1.1, color: 0x9b59b6, xp: 9, mat: 3, ranged: true, shootCd: 2.2 },
    brute:    { hp: 110, speed: 4.8, dmg: 24, size: 1.8, color: 0x8e0000, xp: 18, mat: 5 }
  }[type];

  const geo = new THREE.BoxGeometry(cfg.size * 1.4, cfg.size * 1.6, cfg.size * 1.4);
  const mat = new THREE.MeshStandardMaterial({ color: cfg.color });
  const mesh = new THREE.Mesh(geo, mat);
  mesh.position.set(x, cfg.size * 0.8, z);
  mesh.castShadow = true;
  scene.add(mesh);

  enemies.push({
    mesh,
    type,
    hp: Math.round(cfg.hp * waveMultHp),
    maxHp: Math.round(cfg.hp * waveMultHp),
    speed: cfg.speed + (state.wave * 0.25),
    dmg: Math.round(cfg.dmg * waveMultDmg),
    size: cfg.size,
    xp: cfg.xp,
    mat: cfg.mat,
    ranged: cfg.ranged || false,
    shootTimer: Math.random() * 1.5,
    shootCd: cfg.shootCd || 2.5,
    hitFlash: 0
  });
}

// --- COMBAT & PROJECTILES ---
function fireWeapon(wep, origin) {
  if (enemies.length === 0) return;
  // Find nearest enemy
  let nearest = null;
  let minDist = Infinity;
  for (let e of enemies) {
    const d = origin.distanceTo(e.mesh.position);
    if (d < minDist) { minDist = d; nearest = e; }
  }
  if (!nearest || minDist > 32) return;

  const baseAngle = Math.atan2(nearest.mesh.position.x - origin.x, nearest.mesh.position.z - origin.z);

  for (let i = 0; i < wep.count; i++) {
    const offset = (i - (wep.count - 1) / 2) * wep.spread;
    const angle = baseAngle + offset;

    const geo = new THREE.SphereGeometry(0.25, 6, 6);
    const mat = new THREE.MeshBasicMaterial({ color: wep.color });
    const mesh = new THREE.Mesh(geo, mat);
    mesh.position.set(origin.x, 1.2, origin.z);
    scene.add(mesh);

    const isCrit = Math.random() < state.player.critChance;
    const dmg = wep.dmg * state.player.damageMult * (isCrit ? 1.75 : 1.0);

    playerBullets.push({
      mesh,
      vx: Math.sin(angle) * wep.speed,
      vz: Math.cos(angle) * wep.speed,
      dmg,
      life: 1.4,
      pierce: wep.pierce || 0,
      hitList: []
    });
  }
}

function fireEnemyBullet(enemy) {
  const angle = Math.atan2(playerGroup.position.x - enemy.mesh.position.x, playerGroup.position.z - enemy.mesh.position.z);
  const geo = new THREE.SphereGeometry(0.4, 8, 8);
  const mat = new THREE.MeshBasicMaterial({ color: 0xff0055 });
  const mesh = new THREE.Mesh(geo, mat);
  mesh.position.set(enemy.mesh.position.x, 1.2, enemy.mesh.position.z);
  scene.add(mesh);

  enemyBullets.push({
    mesh,
    vx: Math.sin(angle) * 16,
    vz: Math.cos(angle) * 16,
    dmg: enemy.dmg,
    life: 3.0
  });
}

function spawnDrop(x, z, xp, mat) {
  const geo = new THREE.OctahedronGeometry(0.45);
  const material = new THREE.MeshStandardMaterial({ color: 0x2ed573, emissive: 0x115522 });
  const mesh = new THREE.Mesh(geo, material);
  mesh.position.set(x + (Math.random()-0.5), 0.6, z + (Math.random()-0.5));
  scene.add(mesh);
  drops.push({ mesh, xp, mat });
}

function damagePlayer(rawDmg) {
  if (invulnTimer > 0 || state.gameOver) return;
  // Armor reduces damage, minimum 2 damage always taken
  const mitigated = Math.max(2, Math.round(rawDmg - state.player.armor));
  state.player.hp -= mitigated;
  invulnTimer = 0.35; // Very short invulnerability window!
  spudMat.color.setHex(0xff0000);

  if (state.player.hp <= 0) {
    state.player.hp = 0;
    triggerGameOver();
  }
  updateHUD();
}

// --- LEVEL UP & SHOP UI ---
function checkLevelUp() {
  while (state.xp >= state.xpToNext) {
    state.xp -= state.xpToNext;
    state.level++;
    state.xpToNext = Math.round(state.xpToNext * CONFIG.xpRequirementScale);
    state.pendingLevelUps++;
  }
  if (state.pendingLevelUps > 0 && !state.paused) {
    showLevelUpModal();
  }
}

function showLevelUpModal() {
  state.paused = true;
  const modal = document.getElementById('levelup-modal');
  const container = document.getElementById('levelup-cards');
  container.innerHTML = '';

  const choices = [...NERFED_UPGRADES].sort(() => 0.5 - Math.random()).slice(0, 3);
  choices.forEach(upg => {
    const card = document.createElement('div');
    card.className = 'card';
    card.innerHTML = `
      <h3>${upg.name}</h3>
      <div class="stat-desc">${upg.desc}</div>
      <div class="nerf-note">${upg.nerf}</div>
      <button class="btn btn-buy">SELECT</button>
    `;
    card.onclick = () => {
      upg.apply();
      state.pendingLevelUps--;
      modal.style.display = 'none';
      if (state.pendingLevelUps > 0) {
        showLevelUpModal();
      } else {
        state.paused = false;
        if (state.waveTimer <= 0) openShopModal();
      }
      updateHUD();
    };
    container.appendChild(card);
  });

  modal.style.display = 'flex';
}

function openShopModal() {
  state.paused = true;
  // Clear remaining enemies & bullets
  enemies.forEach(e => scene.remove(e.mesh));
  enemies = [];
  enemyBullets.forEach(b => scene.remove(b.mesh));
  enemyBullets = [];
  playerBullets.forEach(b => scene.remove(b.mesh));
  playerBullets = [];

  // Auto-collect remaining drops
  drops.forEach(d => {
    state.xp += d.xp;
    state.materials += d.mat;
    scene.remove(d.mesh);
  });
  drops = [];

  state.rerollCost = Math.round(CONFIG.rerollBaseCost * Math.pow(1.25, state.wave - 1));
  document.getElementById('next-wave-btn').innerText = `START WAVE ${state.wave + 1}`;
  renderShopItems();
  document.getElementById('shop-modal').style.display = 'flex';
}

function getInflatedPrice(basePrice) {
  return Math.round(basePrice * Math.pow(CONFIG.shopInflation, state.wave - 1));
}

function renderShopItems() {
  document.getElementById('shop-mat-text').innerText = state.materials;
  const rerollBtn = document.getElementById('reroll-btn');
  rerollBtn.innerText = `Reroll Shop (Cost: ${state.rerollCost})`;
  rerollBtn.disabled = state.materials < state.rerollCost;

  const container = document.getElementById('shop-cards');
  container.innerHTML = '';

  const choices = [...SHOP_POOL].sort(() => 0.5 - Math.random()).slice(0, 3);
  choices.forEach(item => {
    const price = getInflatedPrice(item.basePrice);
    const wepFull = item.type === 'weapon' && state.player.weapons.length >= 6;
    const canAfford = state.materials >= price && !wepFull;

    const card = document.createElement('div');
    card.className = 'card';
    card.innerHTML = `
      <h3>${item.name}</h3>
      <div class="stat-desc">${item.desc}</div>
      <div class="nerf-note">10x Inflation Active</div>
      <button class="btn btn-buy" ${!canAfford ? 'disabled' : ''}>
        ${wepFull ? 'SLOTS FULL' : `BUY (${price} MAT)`}
      </button>
    `;
    const btn = card.querySelector('button');
    btn.onclick = () => {
      if (state.materials < price) return;
      state.materials -= price;
      if (item.type === 'weapon') {
        addWeapon(item.id);
      } else {
        item.apply();
      }
      btn.disabled = true;
      btn.innerText = "SOLD OUT";
      document.getElementById('shop-mat-text').innerText = state.materials;
      rerollBtn.disabled = state.materials < state.rerollCost;
      updateHUD();
    };
    container.appendChild(card);
  });
}

document.getElementById('reroll-btn').onclick = () => {
  if (state.materials >= state.rerollCost) {
    state.materials -= state.rerollCost;
    state.rerollCost = Math.round(state.rerollCost * 1.5);
    renderShopItems();
    updateHUD();
  }
};

document.getElementById('next-wave-btn').onclick = () => {
  state.wave++;
  state.waveTimer = CONFIG.baseWaveDuration + (state.wave - 1) * CONFIG.waveTimeScale;
  // Small wave heal (nerfed: only 10% of max HP)
  state.player.hp = Math.min(state.player.maxHp, state.player.hp + Math.ceil(state.player.maxHp * 0.1));
  document.getElementById('shop-modal').style.display = 'none';
  state.paused = false;
  updateHUD();
};

function triggerGameOver() {
  state.gameOver = true;
  state.paused = true;
  document.getElementById('gameover-stats').innerText = `Survived to Wave ${state.wave} | Level ${state.level} | Total Kills: ${state.kills}`;
  document.getElementById('gameover-modal').style.display = 'flex';
}

// --- HUD UPDATE ---
function updateHUD() {
  const hpPct = Math.max(0, (state.player.hp / state.player.maxHp) * 100);
  document.getElementById('hp-bar').style.width = hpPct + '%';
  document.getElementById('hp-text').innerText = `HP: ${state.player.hp}/${state.player.maxHp}`;

  const xpPct = Math.min(100, (state.xp / state.xpToNext) * 100);
  document.getElementById('xp-bar').style.width = xpPct + '%';
  document.getElementById('xp-text').innerText = `LEVEL ${state.level} (${state.xp}/${state.xpToNext} XP)`;

  document.getElementById('mat-text').innerText = state.materials;
  document.getElementById('wave-title').innerText = `WAVE ${state.wave}`;
  document.getElementById('wave-timer').innerText = Math.ceil(state.waveTimer);
  document.getElementById('wep-text').innerText = `${state.player.weapons.length}/6`;
  document.getElementById('alive-text').innerText = enemies.length;
  document.getElementById('kills-text').innerText = state.kills;
  document.getElementById('player-stats-summary').innerText = 
    `DMG: ${Math.round(state.player.damageMult*100)}% | ASPD: ${Math.round(state.player.attackSpeedMult*100)}% | ARM: ${state.player.armor}`;
}

// --- MAIN GAME LOOP ---
const clock = new THREE.Clock();
let invulnTimer = 0;

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.1);

  if (state.paused || state.gameOver) {
    renderer.render(scene, camera);
    return;
  }

  // Invulnerability flash reset
  if (invulnTimer > 0) {
    invulnTimer -= dt;
    if (invulnTimer <= 0) spudMat.color.setHex(0xd4a373);
  }

  // Wave Timer
  state.waveTimer -= dt;
  if (state.waveTimer <= 0) {
    state.waveTimer = 0;
    updateHUD();
    openShopModal();
    return;
  }

  // Player Movement
  let moveX = 0, moveZ = 0;
  if (keys['KeyW'] || keys['ArrowUp']) moveZ -= 1;
  if (keys['KeyS'] || keys['ArrowDown']) moveZ += 1;
  if (keys['KeyA'] || keys['ArrowLeft']) moveX -= 1;
  if (keys['KeyD'] || keys['ArrowRight']) moveX += 1;

  if (moveX !== 0 || moveZ !== 0) {
    const len = Math.hypot(moveX, moveZ);
    moveX /= len; moveZ /= len;
    playerGroup.position.x += moveX * state.player.speed * dt;
    playerGroup.position.z += moveZ * state.player.speed * dt;
    playerGroup.rotation.y = Math.atan2(moveX, moveZ);
  }

  // Clamp Player to Arena
  const limit = CONFIG.arenaSize - 1.5;
  playerGroup.position.x = Math.max(-limit, Math.min(limit, playerGroup.position.x));
  playerGroup.position.z = Math.max(-limit, Math.min(limit, playerGroup.position.z));

  // Camera follow
  camera.position.x = playerGroup.position.x * 0.75;
  camera.position.z = playerGroup.position.z * 0.75 + 26;
  camera.lookAt(playerGroup.position.x * 0.75, 0, playerGroup.position.z * 0.75);

  // Orbit Weapons & Fire
  const time = clock.getElapsedTime();
  state.player.weapons.forEach((w, idx) => {
    const angle = time * 2.5 + (idx / state.player.weapons.length) * Math.PI * 2;
    const mesh = weaponMeshes[idx];
    if (mesh) {
      mesh.position.set(Math.cos(angle) * 1.9, 1.3, Math.sin(angle) * 1.9);
      mesh.rotation.y = -angle;
    }
    w.timer -= dt * state.player.attackSpeedMult;
    if (w.timer <= 0) {
      fireWeapon(w, playerGroup.position);
      w.timer = w.cd;
    }
  });

  // Enemy Spawning (Gets frantic fast!)
  spawnTimer -= dt;
  const currentSpawnInterval = Math.max(0.14, CONFIG.enemySpawnRateBase / (1 + (state.wave - 1) * 0.45));
  if (spawnTimer <= 0) {
    // Spawn bursts on higher waves
    const burst = state.wave >= 3 ? 2 : 1;
    for (let i = 0; i < burst; i++) spawnEnemy();
    spawnTimer = currentSpawnInterval;
  }

  // Update Enemies
  for (let i = enemies.length - 1; i >= 0; i--) {
    const e = enemies[i];
    const dx = playerGroup.position.x - e.mesh.position.x;
    const dz = playerGroup.position.z - e.mesh.position.z;
    const dist = Math.hypot(dx, dz);

    if (dist > 0.1) {
      // Ranged enemies stop at mid-range to shoot
      if (!e.ranged || dist > 16) {
        e.mesh.position.x += (dx / dist) * e.speed * dt;
        e.mesh.position.z += (dz / dist) * e.speed * dt;
      }
      e.mesh.rotation.y = Math.atan2(dx, dz);
    }

    // Ranged Attack
    if (e.ranged) {
      e.shootTimer -= dt;
      if (e.shootTimer <= 0 && dist < 28) {
        fireEnemyBullet(e);
        e.shootTimer = e.shootCd;
      }
    }

    // Contact Damage
    if (dist < e.size + 1.0) {
      damagePlayer(e.dmg);
    }
  }

  // Update Player Bullets
  for (let i = playerBullets.length - 1; i >= 0; i--) {
    const b = playerBullets[i];
    b.mesh.position.x += b.vx * dt;
    b.mesh.position.z += b.vz * dt;
    b.life -= dt;

    let removeBullet = b.life <= 0;

    for (let j = enemies.length - 1; j >= 0; j--) {
      const e = enemies[j];
      if (b.hitList.includes(e)) continue;
      const dist = Math.hypot(b.mesh.position.x - e.mesh.position.x, b.mesh.position.z - e.mesh.position.z);
      if (dist < e.size + 0.5) {
        e.hp -= b.dmg;
        b.hitList.push(e);
        if (e.hp <= 0) {
          spawnDrop(e.mesh.position.x, e.mesh.position.z, e.xp, e.mat);
          scene.remove(e.mesh);
          enemies.splice(j, 1);
          state.kills++;
        }
        if (b.pierce > 0) {
          b.pierce--;
        } else {
          removeBullet = true;
          break;
        }
      }
    }

    if (removeBullet) {
      scene.remove(b.mesh);
      playerBullets.splice(i, 1);
    }
  }

  // Update Enemy Bullets
  for (let i = enemyBullets.length - 1; i >= 0; i--) {
    const b = enemyBullets[i];
    b.mesh.position.x += b.vx * dt;
    b.mesh.position.z += b.vz * dt;
    b.life -= dt;

    const dist = Math.hypot(b.mesh.position.x - playerGroup.position.x, b.mesh.position.z - playerGroup.position.z);
    if (dist < 1.25) {
      damagePlayer(b.dmg);
      scene.remove(b.mesh);
      enemyBullets.splice(i, 1);
      continue;
    }

    if (b.life <= 0) {
      scene.remove(b.mesh);
      enemyBullets.splice(i, 1);
    }
  }

  // Update Drops (Magnet Pickup)
  for (let i = drops.length - 1; i >= 0; i--) {
    const d = drops[i];
    d.mesh.rotation.y += 3 * dt;
    const dx = playerGroup.position.x - d.mesh.position.x;
    const dz = playerGroup.position.z - d.mesh.position.z;
    const dist = Math.hypot(dx, dz);

    if (dist < state.player.pickupRange) {
      d.mesh.position.x += (dx / dist) * 22 * dt;
      d.mesh.position.z += (dz / dist) * 22 * dt;
    }

    if (dist < 1.4) {
      state.xp += d.xp;
      state.materials += d.mat;
      scene.remove(d.mesh);
      drops.splice(i, 1);
      checkLevelUp();
    }
  }

  updateHUD();
  renderer.render(scene, camera);
}

window.addEventListener('resize', () => {
  camera.aspect = window.innerWidth / window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
});

updateHUD();
animate();
</script>
